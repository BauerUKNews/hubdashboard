-- Run once in Supabase > SQL Editor, before publishing the new index.html. -- Moves Hereford & Worcester with its current statuses and notes intact. -- Adds Bristol at every existing AM/PM time, initially PENDING. -- Safe to re-run: existing Bristol entries are left unchanged. -- Runs as one transaction: any error rolls back the whole update. BEGIN; LOCK TABLE public.bulletins IN SHARE ROW EXCLUSIVE MODE; DO $$ BEGIN IF EXISTS ( SELECT 1 FROM public.bulletins old JOIN public.bulletins dest ON dest.key = 'nm:H&W|' || old.slot WHERE old.hub = 'sw' AND old.region = 'H&W' ) THEN RAISE EXCEPTION 'Hereford & Worcester exists in both hubs. Nothing changed; check the duplicate rows before continuing.'; END IF; END $$; -- Replace the original hub/region rule inside the same transaction. -- The table lock prevents other writes until the new rule is in place. ALTER TABLE public.bulletins DROP CONSTRAINT bulletin_region_matches_hub; UPDATE public.bulletins SET hub = 'nm', key = 'nm:H&W|' || slot WHERE hub = 'sw' AND region = 'H&W'; INSERT INTO public.bulletins (key, hub, region, shift, slot, status, note) SELECT 'sw:BRI|' || slot, 'sw', 'BRI', shift, slot, 'PENDING', '' FROM (VALUES ('AM', '0600'), ('AM', '0700'), ('AM', '0800'), ('AM', '0900'), ('AM', '1000'), ('AM', '1100'), ('AM', '1200'), ('AM', '1300'), ('PM', '1400'), ('PM', '1500'), ('PM', '1600'), ('PM', '1700'), ('PM', '1800'), ('PM', '1900') ) AS schedule(shift, slot) ON CONFLICT (key) DO NOTHING; ALTER TABLE public.bulletins ADD CONSTRAINT bulletin_region_matches_hub CHECK ( (hub = 'sw' AND region IN ('BRI','CWL','C&W','DEV','DOR','GLO','SOM','SWI','WST','WIL')) OR (hub = 'se' AND region IN ('BRK','BBH','CAM','ESX','NHT','NFK','OXF','SUF','SUR','SUS')) OR (hub = 'nm' AND region IN ('BCS','CMB','EYRK','H&W','LAN','LIN','NYRK','PSR','S&C')) ); DO $$ BEGIN IF EXISTS (SELECT 1 FROM public.bulletins WHERE hub = 'sw' AND region = 'H&W') OR (SELECT count(*) FROM public.bulletins WHERE hub = 'nm' AND region = 'H&W') <> 14 OR (SELECT count(*) FROM public.bulletins WHERE hub = 'sw' AND region = 'BRI') <> 14 THEN RAISE EXCEPTION 'Bulletin counts did not match the expected 14 time slots. Nothing changed.'; END IF; END $$; COMMIT; -- Expected result: nm / H&W / 14, and sw / BRI / 14. SELECT hub, region, count(*) AS bulletin_count FROM public.bulletins WHERE (hub = 'nm' AND region = 'H&W') OR (hub = 'sw' AND region = 'BRI') GROUP BY hub, region ORDER BY hub, region;
